function guard() {
  if (process.env.REVIEW_VALIDATION_ENABLED !== 'true' || !['https://precious-pheasant-866.convex.cloud','https://usable-snail-909.convex.cloud'].includes(process.env.CONVEX_CLOUD_URL ?? '')) throw new Error('Named synthetic preview only')
}
/** Force expiry without a 20-minute wait; refuses non-synthetic owners. */
export const expireAttempts = internalMutation({
  args: { tokenHash: v.string() }, returns: v.null(), handler: async (ctx, { tokenHash }) => {
    guard()
    const grant = await ctx.db.query('trmnlGrants').withIndex('by_tokenHash', q => q.eq('tokenHash', tokenHash)).unique()
    const owner = grant ? await ctx.db.get(grant.userId) : null
    if (!grant || !owner?.tokenIdentifier.startsWith('https://review.invalid|user_synthetic')) throw new Error('Synthetic owner required')
    for (const attempt of await ctx.db.query('trmnlInstallAttempts').withIndex('by_grantId_and_state', q => q.eq('grantId', grant._id).eq('state', 'pending')).take(10)) {
      await ctx.db.patch(attempt._id, { expiresAt: Date.now() - 1 })
    }
    return null
  },
})
